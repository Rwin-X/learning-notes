3. File & Folder Automation

List, filter, and rename files

from pathlib import Path

folder = Path("./downloads")

# List all PDF files
pdf_files = list(folder.glob("*.pdf"))
print(f"Found {len(pdf_files)} PDFs")

# Recursive search through subfolders
all_images = list(folder.rglob("*.jpg")) + list(folder.rglob("*.png"))

# Rename every .txt file to add a "processed_" prefix
for file in folder.glob("*.txt"):
    new_name = file.with_name(f"processed_{file.name}")
    file.rename(new_name)
    print(f"Renamed {file.name} -> {new_name.name}")

Organize files into folders by extension

from pathlib import Path
import shutil

source = Path("./Downloads")

CATEGORY_MAP = {
    ".pdf": "PDFs",
    ".jpg": "Images", ".png": "Images", ".gif": "Images",
    ".mp4": "Videos", ".mkv": "Videos",
    ".zip": "Archives", ".tar": "Archives", ".gz": "Archives",
    ".py": "Code", ".js": "Code",
}

def organize_downloads(source: Path):
    for file in source.iterdir():
        if file.is_dir():
            continue
        category = CATEGORY_MAP.get(file.suffix.lower(), "Other")
        dest_folder = source / category
        dest_folder.mkdir(exist_ok=True)
        shutil.move(str(file), dest_folder / file.name)
        print(f"{file.name} -> {category}/")

organize_downloads(source)

Find and delete duplicate files by hash

import hashlib
from pathlib import Path
from collections import defaultdict

def file_hash(path: Path, chunk_size=8192):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(chunk_size), b""):
            h.update(chunk)
    return h.hexdigest()

def find_duplicates(folder: Path):
    hashes = defaultdict(list)
    for file in folder.rglob("*"):
        if file.is_file():
            hashes[file_hash(file)].append(file)
    return {h: paths for h, paths in hashes.items() if len(paths) > 1}

duplicates = find_duplicates(Path("./photos"))
for file_hash_val, paths in duplicates.items():
    print("Duplicate set:")
    for p in paths:
        print("  ", p)

5. Copy, Move, Archive with shutil

import shutil
from pathlib import Path

# Copy a single file
shutil.copy("source.txt", "backup/source.txt")
shutil.copy2("source.txt", "backup/source.txt")  # preserves metadata (timestamps)

# Copy an entire directory tree
shutil.copytree("project", "project_backup", dirs_exist_ok=True)

# Move (works across filesystems, unlike os.rename)
shutil.move("temp/report.pdf", "archive/report.pdf")

# Delete a directory tree (careful!)
shutil.rmtree("old_build", ignore_errors=True)

# Create a zip archive
shutil.make_archive("backup_2026_08_16", "zip", root_dir="project")

# Extract a zip archive
shutil.unpack_archive("backup_2026_08_16.zip", extract_dir="restored")

# Check free disk space before a big operation
total, used, free = shutil.disk_usage(".")
print(f"Free: {free // (2**30)} GB")
shutil.rmtree deletes permanently — no recycle bin. Always double-check the path (ideally with a dry-run print) before calling it in an automated script.

6. Running System Commands

import subprocess

# Run a command and capture output (preferred over os.system)
result = subprocess.run(
    ["ping", "-c", "4", "example.com"],
    capture_output=True,
    text=True,
    timeout=15,
)
print("stdout:", result.stdout)
print("exit code:", result.returncode)

# Raise an exception automatically on non-zero exit
try:
    subprocess.run(["git", "pull"], check=True, cwd="/home/user/project")
except subprocess.CalledProcessError as e:
    print(f"git pull failed with code {e.returncode}")

# Run a shell pipeline (use with caution - shell=True can be a security risk
# if any part of the command comes from untrusted/user input)
result = subprocess.run("ls -la | grep .py", shell=True, capture_output=True, text=True)
print(result.stdout)
Avoid shell=True with untrusted input. Building a command string from user-supplied data and running it with shell=True is a classic command-injection vulnerability. Pass a list of arguments instead whenever possible.

7. Watching Folders for Changes

import time
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class NewFileHandler(FileSystemEventHandler):
    def on_created(self, event):
        if not event.is_directory:
            print(f"New file detected: {event.src_path}")
            # e.g. trigger processing, move the file, notify someone...

    def on_modified(self, event):
        if not event.is_directory:
            print(f"Modified: {event.src_path}")

    def on_deleted(self, event):
        print(f"Deleted: {event.src_path}")

path_to_watch = "./incoming"
event_handler = NewFileHandler()
observer = Observer()
observer.schedule(event_handler, path_to_watch, recursive=True)
observer.start()

print(f"Watching {path_to_watch} for changes... (Ctrl+C to stop)")
try:
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    observer.stop()
observer.join()

8. In-Process Scheduling with schedule

import schedule
import time

def backup_job():
    print("Running daily backup...")

def check_prices():
    print("Checking prices...")

# Different schedule patterns
schedule.every().day.at("02:00").do(backup_job)
schedule.every(10).minutes.do(check_prices)
schedule.every().monday.at("09:00").do(lambda: print("Weekly report time"))
schedule.every().hour.do(lambda: print("Hourly check-in"))

print("Scheduler running. Press Ctrl+C to stop.")
while True:
    schedule.run_pending()
    time.sleep(1)
Note: schedule only runs while your Python process is alive. For "always runs, even after reboot" scheduling, use the OS scheduler (cron / Task Scheduler) instead — see the next section.

9. OS-Level Scheduling: cron & Task Scheduler

Linux/macOS — cron

# Edit your crontab
crontab -e

# Run a script every day at 2:00 AM
0 2 * * * /usr/bin/python3 /home/rwin/scripts/backup.py >> /home/rwin/logs/backup.log 2>&1

# Run every 15 minutes
*/15 * * * * /usr/bin/python3 /home/rwin/scripts/check_prices.py

# Run every Monday at 9 AM
0 9 * * 1 /usr/bin/python3 /home/rwin/scripts/weekly_report.py

Windows — Task Scheduler (PowerShell)

# Create a daily scheduled task at 2:00 AM
$action = New-ScheduledTaskAction -Execute "python.exe" -Argument "C:\scripts\backup.py"
$trigger = New-ScheduledTaskTrigger -Daily -At 2am
Register-ScheduledTask -TaskName "DailyBackup" -Action $action -Trigger $trigger
Always use absolute paths for the Python interpreter, script, and any files it touches in cron/Task Scheduler jobs — the working directory and PATH are often different from your interactive shell.

10. Advanced Scheduling with APScheduler

from apscheduler.schedulers.blocking import BlockingScheduler
from datetime import datetime

scheduler = BlockingScheduler()

def job_interval():
    print(f"[{datetime.now()}] Running every 30 seconds")

def job_cron():
    print("Running at 08:30 on weekdays")

# Interval trigger
scheduler.add_job(job_interval, "interval", seconds=30)

# Cron-style trigger (weekdays at 8:30 AM)
scheduler.add_job(job_cron, "cron", day_of_week="mon-fri", hour=8, minute=30)

# One-time run at a specific date/time
scheduler.add_job(
    lambda: print("One-time reminder!"),
    "date",
    run_date="2026-12-31 23:59:00",
)

print("Starting APScheduler...")
scheduler.start()

11. Web Scraping requests + BeautifulSoup

For static HTML pages that don't require JavaScript rendering, this combo is fast and lightweight.

import requests
from bs4 import BeautifulSoup

headers = {"User-Agent": "Mozilla/5.0 (automation-script/1.0)"}
response = requests.get("https://example.com/articles", headers=headers, timeout=10)
response.raise_for_status()

soup = BeautifulSoup(response.text, "lxml")

# Find all article titles and links
for article in soup.select("article.post"):
    title = article.select_one("h2.title")
    link = article.select_one("a")
    if title and link:
        print(title.get_text(strip=True), "->", link["href"])

# Extracting a table into rows of dicts
table = soup.find("table", {"id": "price-table"})
headers_row = [th.get_text(strip=True) for th in table.find_all("th")]
rows = []
for tr in table.find_all("tr")[1:]:
    cells = [td.get_text(strip=True) for td in tr.find_all("td")]
    rows.append(dict(zip(headers_row, cells)))

print(rows[:3])

Polite scraping: throttling + retries

import time
import requests
from bs4 import BeautifulSoup

def scrape_pages(urls, delay=1.5):
    session = requests.Session()
    session.headers.update({"User-Agent": "automation-script/1.0 (contact: you@example.com)"})

    results = []
    for url in urls:
        try:
            response = session.get(url, timeout=10)
            response.raise_for_status()
            soup = BeautifulSoup(response.text, "lxml")
            results.append({"url": url, "title": soup.title.get_text(strip=True) if soup.title else None})
        except requests.exceptions.RequestException as e:
            print(f"Failed {url}: {e}")
        time.sleep(delay)   # be polite - don't hammer the server

    return results
Check robots.txt and terms of service before scraping any site, throttle your requests, and identify yourself with a real User-Agent. Never scrape content behind a login without authorization.

12. Browser Automation Selenium

Selenium drives a real browser — needed for JavaScript-heavy sites, logins, clicking, and form-filling.

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.chrome.options import Options

options = Options()
options.add_argument("--headless=new")   # run without opening a visible window
options.add_argument("--window-size=1920,1080")

driver = webdriver.Chrome(options=options)

try:
    driver.get("https://example.com/login")

    # Wait for an element instead of a fixed sleep() - much more reliable
    email_field = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.NAME, "email"))
    )
    email_field.send_keys("user@example.com")
    driver.find_element(By.NAME, "password").send_keys("secret123")
    driver.find_element(By.CSS_SELECTOR, "button[type='submit']").click()

    WebDriverWait(driver, 10).until(EC.url_contains("/dashboard"))
    print("Logged in:", driver.current_url)

    # Extract data from the rendered page
    items = driver.find_elements(By.CSS_SELECTOR, ".order-row")
    for item in items:
        print(item.text)

    driver.save_screenshot("dashboard.png")

finally:
    driver.quit()

13. Browser Automation Playwright

Playwright is the modern alternative to Selenium — faster, built-in auto-waiting, and a cleaner API. Recommended for new projects.

from playwright.sync_api import sync_playwright

with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page()

    page.goto("https://example.com/login")
    page.fill("input[name='email']", "user@example.com")
    page.fill("input[name='password']", "secret123")
    page.click("button[type='submit']")

    # Playwright auto-waits for navigation and elements - no manual sleeps needed
    page.wait_for_url("**/dashboard")
    print("Logged in:", page.url)

    # Extract text from multiple elements
    rows = page.locator(".order-row").all_text_contents()
    for row in rows:
        print(row)

    page.screenshot(path="dashboard.png", full_page=True)
    browser.close()

Async Playwright for concurrent browser tasks

import asyncio
from playwright.async_api import async_playwright

async def check_page_title(playwright, url):
    browser = await playwright.chromium.launch(headless=True)
    page = await browser.new_page()
    await page.goto(url, timeout=15000)
    title = await page.title()
    await browser.close()
    return url, title

async def main():
    urls = ["https://example.com", "https://python.org", "https://github.com"]
    async with async_playwright() as p:
        results = await asyncio.gather(*[check_page_title(p, u) for u in urls])
    for url, title in results:
        print(f"{url} -> {title}")

asyncio.run(main())

Downloading files & handling dialogs

with sync_playwright() as p:
    browser = p.chromium.launch()
    page = browser.new_page()
    page.goto("https://example.com/reports")

    # Handle a JS confirm/alert dialog automatically
    page.on("dialog", lambda dialog: dialog.accept())

    # Capture a file download triggered by clicking a button
    with page.expect_download() as download_info:
        page.click("text=Download CSV")
    download = download_info.value
    download.save_as(f"./downloads/{download.suggested_filename}")

    browser.close()

17. Email Automation

Sending email with attachments

import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders
import os

def send_email_with_attachment(to_addr, subject, body, attachment_path=None):
    sender = os.environ["SMTP_USER"]
    password = os.environ["SMTP_PASSWORD"]   # never hardcode this

    msg = MIMEMultipart()
    msg["From"] = sender
    msg["To"] = to_addr
    msg["Subject"] = subject
    msg.attach(MIMEText(body, "plain"))

    if attachment_path:
        with open(attachment_path, "rb") as f:
            part = MIMEBase("application", "octet-stream")
            part.set_payload(f.read())
        encoders.encode_base64(part)
        part.add_header(
            "Content-Disposition",
            f"attachment; filename={os.path.basename(attachment_path)}",
        )
        msg.attach(part)

    with smtplib.SMTP("smtp.gmail.com", 587) as server:
        server.starttls()
        server.login(sender, password)
        server.send_message(msg)

send_email_with_attachment(
    "team@example.com",
    "Daily Report - 2026-08-16",
    "Attached is today's automated report.",
    "report.pdf",
)

Reading unread emails (IMAP)

import imaplib
import email
import os

mail = imaplib.IMAP4_SSL("imap.gmail.com")
mail.login(os.environ["IMAP_USER"], os.environ["IMAP_PASSWORD"])
mail.select("inbox")

status, message_ids = mail.search(None, "UNSEEN")

for msg_id in message_ids[0].split():
    status, data = mail.fetch(msg_id, "(RFC822)")
    msg = email.message_from_bytes(data[0][1])
    print("From:", msg["From"])
    print("Subject:", msg["Subject"])

    if msg.is_multipart():
        for part in msg.walk():
            if part.get_content_type() == "text/plain":
                print(part.get_payload(decode=True).decode(errors="ignore"))

mail.logout()
Use an app password, never your real account password, and store it in an environment variable — not in the script. Most providers (Gmail, Outlook) require this for SMTP/IMAP access, and it can be revoked independently of your login credentials.

18. Desktop GUI Automation pyautogui

For controlling legacy or non-scriptable desktop applications by simulating real mouse/keyboard input.

import pyautogui
import time

pyautogui.FAILSAFE = True   # move mouse to a screen corner to abort - keep this on

# Get current mouse position and screen size
print(pyautogui.position())
print(pyautogui.size())

# Move and click
pyautogui.moveTo(500, 300, duration=0.5)
pyautogui.click()
pyautogui.doubleClick(800, 400)
pyautogui.rightClick()

# Type text and send key combinations
pyautogui.write("Hello from automation", interval=0.05)
pyautogui.hotkey("ctrl", "s")   # save
pyautogui.press("enter")

# Find an image on screen and click its center (great for automating
# apps with no accessible API - match a button screenshot you saved earlier)
location = pyautogui.locateCenterOnScreen("submit_button.png", confidence=0.8)
if location:
    pyautogui.click(location)
else:
    print("Button not found on screen")

# Take a screenshot
screenshot = pyautogui.screenshot()
screenshot.save("screen.png")
GUI automation is brittle — it breaks if the window moves, resizes, or the OS theme changes. Prefer an API, CLI, or browser automation library whenever one exists; reach for pyautogui only when there's truly no programmatic alternative.

21. Real-World Automation Patterns

Pattern: Nightly backup script (files + logging + notification)

import shutil
import logging
from pathlib import Path
from datetime import datetime

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s")
logger = logging.getLogger("backup")

def run_backup(source: Path, backup_root: Path, keep_last=7):
    timestamp = datetime.now().strftime("%Y-%m-%d_%H%M%S")
    dest = backup_root / f"backup_{timestamp}"

    logger.info(f"Backing up {source} -> {dest}")
    shutil.copytree(source, dest)

    # prune old backups, keep only the most recent N
    backups = sorted(backup_root.glob("backup_*"), reverse=True)
    for old_backup in backups[keep_last:]:
        logger.info(f"Removing old backup: {old_backup}")
        shutil.rmtree(old_backup)

    logger.info("Backup complete")

if __name__ == "__main__":
    run_backup(Path("/home/rwin/projects"), Path("/mnt/backups"))

Pattern: Price/availability monitor with alert

import time
import requests
from bs4 import BeautifulSoup

def check_price(url, selector, target_price):
    response = requests.get(url, headers={"User-Agent": "price-monitor/1.0"}, timeout=10)
    soup = BeautifulSoup(response.text, "lxml")
    price_text = soup.select_one(selector).get_text(strip=True)
    price = float(price_text.replace("$", "").replace(",", ""))

    if price <= target_price:
        notify_slack(SLACK_WEBHOOK, f"🔥 Price dropped to ${price}! {url}")
        return True
    print(f"Current price: ${price} (target: ${target_price})")
    return False

while not check_price("https://example.com/product/123", ".price", 49.99):
    time.sleep(1800)   # check every 30 minutes

22. Security & Safety Notes

Never hardcode credentials. Use environment variables or a secrets manager for SMTP passwords, API keys, and login credentials used by automation scripts.
Dry-run destructive scripts first. Any script that deletes, moves, or overwrites files should support a --dry-run flag that only prints what it would do.
Avoid shell=True and os.system() with dynamic input. Use subprocess.run([...]) with a list of arguments to prevent command injection.
Rate-limit and identify scrapers. Add delays between requests and a real User-Agent so you don't get blocked — or accidentally DoS a small site.
Keep pyautogui's FAILSAFE on. It lets you abort a runaway GUI-automation script instantly by yanking the mouse to a screen corner.
Log everything a scheduled job does. Unattended scripts fail silently otherwise — a log file (and a notification on failure) is how you find out.
