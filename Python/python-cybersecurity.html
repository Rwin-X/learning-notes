01 / NETWORK

Port Scanning

socket 
import socket

def scan_port(host, port, timeout=0.5):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    result = sock.connect_ex((host, port))
    sock.close()
    return result == 0

target = "127.0.0.1"
for port in [22, 80, 443, 3306]:
    if scan_port(target, port):
        print(f"[OPEN] {target}:{port}")
connect_ex() avoids exceptions on closed ports — returns 0 on success.
02 / NETWORK

Packet Sniffing

scapy 
from scapy.all import sniff, IP, TCP

def handle_packet(pkt):
    if pkt.haslayer(IP) and pkt.haslayer(TCP):
        src = pkt[IP].src
        dst = pkt[IP].dst
        print(f"{src} -> {dst} : port {pkt[TCP].dport}")

sniff(filter="tcp", prn=handle_packet, count=10)
Requires elevated privileges (root / Administrator) on most OSes.
03 / CRYPTO

Password Hashing

hashlib 
import hashlib, os

def hash_password(password: str):
    salt = os.urandom(16)
    digest = hashlib.pbkdf2_hmac(
        "sha256", password.encode(), salt, 200_000
    )
    return salt.hex(), digest.hex()

salt, hashed = hash_password("correct-horse-battery-staple")
print(f"salt={salt}\nhash={hashed}")
200k PBKDF2 iterations is a reasonable 2026 baseline; prefer Argon2id where available.
04 / CRYPTO

File Encryption

cryptography 
from cryptography.fernet import Fernet

key = Fernet.generate_key()
cipher = Fernet(key)

with open("secret.txt", "rb") as f:
    encrypted = cipher.encrypt(f.read())

with open("secret.enc", "wb") as f:
    f.write(encrypted)

# Decrypt: cipher.decrypt(encrypted)
Fernet = AES-128-CBC + HMAC, authenticated symmetric encryption.
05 / FORENSICS

Log Parsing

re 
import re

FAILED_LOGIN = re.compile(
    r"Failed password for (\S+) from (\d+\.\d+\.\d+\.\d+)"
)

with open("auth.log") as f:
    for line in f:
        match = FAILED_LOGIN.search(line)
        if match:
            user, ip = match.groups()
            print(f"Failed login: user={user} ip={ip}")
Same pattern generalizes to any IOC extraction from flat log files.
06 / WEB

Header Recon

requests 
import requests

def check_security_headers(url):
    resp = requests.get(url, timeout=5)
    headers_to_check = [
        "Content-Security-Policy",
        "X-Frame-Options",
        "Strict-Transport-Security",
    ]
    for h in headers_to_check:
        status = "present" if h in resp.headers else "MISSING"
        print(f"{h}: {status}")

check_security_headers("https://example.com")
Quick baseline check for missing hardening headers on any target.
