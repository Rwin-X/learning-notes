socket fundamentals socket

every network tool descends from this. AF_INET = IPv4, SOCK_STREAM = TCP, SOCK_DGRAM = UDP.

socket_basics.pySTDLIB
import socket

# create socket: (address_family, socket_type)
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)  # TCP
u = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)   # UDP
r = socket.socket(socket.AF_INET, socket.SOCK_RAW, socket.IPPROTO_TCP)  # RAW - needs root

# options
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.settimeout(5.0)

# resolve hostname
ip = socket.gethostbyname('example.com')
hostname = socket.gethostbyaddr('93.184.216.34')

# local machine info
local_ip = socket.gethostbyname(socket.gethostname())

tcp client / server socket

blocking TCP. bind → listen → accept loop on server; connect → send/recv on client.

tcp_server.pySERVER
import socket

HOST, PORT = '0.0.0.0', 9999

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as srv:
    srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    srv.bind((HOST, PORT))
    srv.listen(5)
    print(f"[+] listening on {PORT}")

    while True:
        conn, addr = srv.accept()
        with conn:
            print(f"[+] conn from {addr}")
            while data := conn.recv(1024):
                conn.sendall(data)  # echo
tcp_client.pyCLIENT
import socket

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
    s.settimeout(3)
    s.connect(('127.0.0.1', 9999))
    s.sendall(b"hello server")
    resp = s.recv(4096)
    print(resp.decode())
production servers: use socketserver.ThreadingTCPServer or asyncio instead of a manual accept loop.

udp sockets socket

connectionless. no accept()/listen() — just bind + recvfrom/sendto.

udp_server.pySERVER
import socket
s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
s.bind(('0.0.0.0', 9000))
while True:
    data, addr = s.recvfrom(1024)
    print(f"{addr}: {data}")
    s.sendto(b"ack", addr)

asyncio networking asyncio

non-blocking, coroutine-based — scales to thousands of concurrent connections.

async_server.pyASYNCIO
import asyncio

async def handle(reader, writer):
    addr = writer.get_extra_info('peername')
    data = await reader.read(1024)
    print(f"{addr}: {data.decode()}")
    writer.write(data)  # echo
    await writer.drain()
    writer.close()

async def main():
    srv = await asyncio.start_server(handle, '0.0.0.0', 8888)
    async with srv:
        await srv.serve_forever()

asyncio.run(main())

# --- client ---
async def client():
    reader, writer = await asyncio.open_connection('127.0.0.1', 8888)
    writer.write(b'hi')
    await writer.drain()
    data = await reader.read(100)
    writer.close()
    await writer.wait_closed()
use asyncio.gather() to fan out concurrent scans/probes — this is your engine for high-throughput tooling (port scanners, OSINT checkers).

httpx_async.pyASYNC
import httpx, asyncio

async def fetch(client, url):
    r = await client.get(url, timeout=5)
    return r.status_code, len(r.content)

async def main(urls):
    async with httpx.AsyncClient() as c:
        tasks = [fetch(c, u) for u in urls]
        return await asyncio.gather(*tasks, return_exceptions=True)

asyncio.run(main(urls))

class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_response(200)
        self.send_header('Content-Type', 'text/plain')

scapy — packet crafting scapy

construct/send/sniff raw packets at any layer. requires root/CAP_NET_RAW. core of PacketForge/TraceForge-style tools.

scapy_craft.pyROOT
from scapy.all import IP, TCP, UDP, ICMP, sr1, send, sniff, Raw

# build layered packet: IP / TCP
pkt = IP(dst="192.168.1.1")/TCP(dport=80, flags="S")
resp = sr1(pkt, timeout=2, verbose=0)   # send + receive 1 reply

if resp and resp.haslayer(TCP):
    flags = resp[TCP].flags
    print("open" if flags == 0x12 else "closed")  # SYN-ACK vs RST

# ICMP ping
icmp_resp = sr1(IP(dst="8.8.8.8")/ICMP(), timeout=2)

# custom payload
pkt = IP(dst=target)/UDP(dport=53)/Raw(load=b"\\xaa\\xbb")
send(pkt, verbose=0)  # fire-and-forget, no response expected

# TTL manipulation (traceroute logic)
for ttl in range(1, 30):
    p = sr1(IP(dst=target, ttl=ttl)/ICMP(), timeout=1, verbose=0)
    if p is None: continue
    print(ttl, p.src)
    if p.src == target: break
raw packet ops require elevated privileges. run with sudo or grant CAP_NET_RAW via setcap.

raw sockets (no scapy) socket + struct

manual packet construction for when you need zero dependencies or full control.

raw_icmp.pyROOT
import socket, struct, os

def checksum(data):
    if len(data) % 2: data += b'\\0'
    s = sum(struct.unpack(f"!{len(data)//2}H", data))
    s = (s >> 16) + (s & 0xffff)
    return ~s & 0xffff

def build_icmp_echo(id_=1, seq=1):
    header = struct.pack('!BBHHH', 8, 0, 0, id_, seq)  # type=8 echo request
    payload = b'devforge-ping'
    cksum = checksum(header + payload)
    header = struct.pack('!BBHHH', 8, 0, cksum, id_, seq)
    return header + payload

s = socket.socket(socket.AF_INET, socket.SOCK_RAW, socket.IPPROTO_ICMP)
s.sendto(build_icmp_echo(), ('8.8.8.8', 0))
print(s.recvfrom(1024))

packet sniffing scapy

capture and filter live traffic. BPF filter syntax = same as tcpdump.

sniffer.pyROOT
from scapy.all import sniff, IP, TCP, Ether

def on_packet(pkt):
    if pkt.haslayer(IP):
        src, dst = pkt[IP].src, pkt[IP].dst
        proto = "TCP" if pkt.haslayer(TCP) else "?"
        print(f"{src} -> {dst} [{proto}]")

sniff(
    filter="tcp port 80 or tcp port 443",   # BPF syntax
    prn=on_packet,
    store=0,                                # don't hold packets in memory
    iface="eth0",                            # None = all interfaces
    count=100,                               # 0 = infinite
    timeout=30
)

port scanner socket + threading

TCP connect scan — no root needed. threaded for speed.

portscan.pyTOOL
import socket
from concurrent.futures import ThreadPoolExecutor

def scan_port(host, port, timeout=0.5):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.settimeout(timeout)
        if s.connect_ex((host, port)) == 0:
            try:
                banner = s.recv(128).decode(errors='ignore').strip()
            except Exception:
                banner = ''
            return port, banner
    return None

def scan_range(host, ports, workers=200):
    open_ports = []
    with ThreadPoolExecutor(max_workers=workers) as ex:
        futures = {ex.submit(scan_port, host, p): p for p in ports}
        for f in futures:
            result = f.result()
            if result: open_ports.append(result)
    return sorted(open_ports)

print(scan_range("scanme.nmap.org", range(1, 1025)))
for stealth (SYN) scans use scapy: send SYN, check for SYN-ACK, send RST manually — never completes the handshake, doesn't hit application logs as easily.

ssl / tls ssl

wrap sockets in TLS, inspect certs — core of cert-checking and MITM-detection tooling.

ssl_inspect.pySTDLIB
import ssl, socket

def get_cert_info(host, port=443):
    ctx = ssl.create_default_context()
    with socket.create_connection((host, port), timeout=5) as sock:
        with ctx.wrap_socket(sock, server_hostname=host) as ssock:
            cert = ssock.getpeercert()
            print("subject:", cert['subject'])
            print("expires:", cert['notAfter'])
            print("cipher:", ssock.cipher())
            print("tls ver:", ssock.version())

# server-side (self-signed for testing)
ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
ctx.load_cert_chain('cert.pem', 'key.pem')
with socket.socket() as sock:
    sock.bind(('0.0.0.0', 8443))
    sock.listen(1)
    with ctx.wrap_socket(sock, server_side=True) as ssock:
        conn, addr = ssock.accept()

error handling essentials

exceptions.pyPATTERN
import socket

try:
    s = socket.create_connection((host, port), timeout=3)
except socket.timeout:
    print("timed out")
except ConnectionRefusedError:
    print("port closed / refused")
except socket.gaierror:
    print("dns resolution failed")
except OSError as e:
    print(f"os error: {e}")  # e.g. network unreachable
